{{- /* Responsive previews keep the untouched original available to the lightbox. Context = publication Page. */ -}} {{- $pub := . }} {{- $isHidden := ($pub.Param "cover.hiddenInList") | default ($pub.Param "cover.hidden") | default false }} {{- if not $isHidden }} {{- with $pub.Params.cover.image }} {{- $path := . }} {{- $img := "" }} {{- $original := "" }} {{- $srcset := "" }} {{- $width := 0 }} {{- $height := 0 }} {{- if or (eq $pub.Params.cover.relative false) (urls.Parse $path).IsAbs (strings.HasPrefix $path "/") }} {{- $img = $path | absURL }} {{- else }} {{- $res := $pub.Resources.Get $path }} {{- if $res }} {{- $preview := partial "responsive-image.html" (dict "resource" $res "widths" (slice 180 360 720)) }} {{- $img = $preview.src }} {{- $original = $preview.original }} {{- $srcset = $preview.srcset }} {{- $width = $preview.width }} {{- $height = $preview.height }} {{- else }} {{- $img = path.Join $pub.RelPermalink $path | absURL }} {{- end }} {{- end }} {{- end }} {{- end }}
